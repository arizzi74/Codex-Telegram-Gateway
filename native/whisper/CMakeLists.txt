cmake_minimum_required(VERSION 3.16)
project(codex_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED WHISPER_SOURCE_DIR)
  message(FATAL_ERROR "WHISPER_SOURCE_DIR must point to the verified pinned source")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
foreach(feature METAL ACCELERATE BLAS CUDA VULKAN OPENMP RPC LLAMAFILE CPU_KLEIDIAI SSE42 AVX AVX2 AVX_VNNI BMI2 FMA F16C AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
add_executable(codex-whisper main.cpp)
target_link_libraries(codex-whisper PRIVATE whisper)
target_compile_definitions(codex-whisper PRIVATE CODEX_WHISPER_VERSION="${CODEX_WHISPER_VERSION}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(codex-whisper PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(protocol-test protocol_test.cpp)
add_test(NAME protocol COMMAND protocol-test)
