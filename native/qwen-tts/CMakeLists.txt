cmake_minimum_required(VERSION 3.16)
project(codex_qwen_tts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED QWEN_TTS_SOURCE_DIR)
  message(FATAL_ERROR "QWEN_TTS_SOURCE_DIR must point to the verified, patched pinned source")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
foreach(feature TESTS TOOLS EXAMPLES SERVER APP UI)
  set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
foreach(feature METAL ACCELERATE BLAS CUDA VULKAN OPENMP RPC LLAMAFILE CPU_KLEIDIAI SSE42 AVX AVX2 AVX_VNNI BMI2 FMA F16C AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
option(QWEN_TTS_OPTIMIZED "Build a helper selected only after CPU capability checks" OFF)
if(QWEN_TTS_OPTIMIZED)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    set(GGML_CPU_ARM_ARCH "armv8-a+dotprod+fp16" CACHE STRING "" FORCE)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    foreach(feature SSE42 AVX AVX2 BMI2 FMA F16C)
      set(GGML_${feature} ON CACHE BOOL "" FORCE)
    endforeach()
  else()
    message(FATAL_ERROR "Unsupported optimized CPU architecture")
  endif()
else()
  set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
add_subdirectory("${QWEN_TTS_SOURCE_DIR}" llama)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/stock-giovanni.wav" stock_wav HEX)
string(REGEX REPLACE "(..)" "0x\\1," stock_wav "${stock_wav}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stock-voice.h" "#pragma once\nstatic const unsigned char stock_voice[] = {${stock_wav}};\n")
add_executable(codex-qwen-tts main.cpp)
target_include_directories(codex-qwen-tts PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(codex-qwen-tts PRIVATE llama mtmd vendor::nlohmann)
target_compile_definitions(codex-qwen-tts PRIVATE CODEX_QWEN_TTS_VERSION="${CODEX_QWEN_TTS_VERSION}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(codex-qwen-tts PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(qwen-protocol-test protocol_test.cpp)
target_link_libraries(qwen-protocol-test PRIVATE vendor::nlohmann)
add_test(NAME qwen-protocol COMMAND qwen-protocol-test)
