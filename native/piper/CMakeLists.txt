cmake_minimum_required(VERSION 3.16)
project(codex_piper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PIPER_SOURCE_DIR OR NOT DEFINED PIPER_RUNTIME_DIR)
  message(FATAL_ERROR "Verified pinned Sherpa source and shared CPU runtime directories are required")
endif()
find_library(SHERPA_C_API NAMES sherpa-onnx-c-api PATHS "${PIPER_RUNTIME_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(codex-piper main.cpp)
target_include_directories(codex-piper PRIVATE "${PIPER_SOURCE_DIR}")
target_link_libraries(codex-piper PRIVATE "${SHERPA_C_API}")
target_compile_definitions(codex-piper PRIVATE CODEX_PIPER_VERSION="${CODEX_PIPER_VERSION}")
target_compile_options(codex-piper PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
  set_target_properties(codex-piper PROPERTIES INSTALL_RPATH "@loader_path/lib" BUILD_WITH_INSTALL_RPATH TRUE)
else()
  # DT_RPATH is intentional: the bundled Sherpa library in turn needs ONNX.
  target_link_options(codex-piper PRIVATE "LINKER:--disable-new-dtags")
  set_target_properties(codex-piper PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
enable_testing()
add_executable(protocol-test protocol_test.cpp)
add_test(NAME protocol COMMAND protocol-test)
add_executable(resample-test resample_test.cpp)
target_include_directories(resample-test PRIVATE "${PIPER_SOURCE_DIR}")
target_link_libraries(resample-test PRIVATE "${SHERPA_C_API}")
add_test(NAME sample-rate-conversion COMMAND resample-test)
